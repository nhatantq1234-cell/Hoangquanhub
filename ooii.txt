local function PlaySound(id)
    task.spawn(function()
        pcall(function()
            local s = Instance.new("Sound")
            s.SoundId = "rbxassetid://" .. tostring(id)
            s.Volume = 1
            s.Parent = workspace 
            s:Play()
            s.Ended:Connect(function() s:Destroy() end)
        end)
    end)
end
