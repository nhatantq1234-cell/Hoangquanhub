-- Tabs
local tabs = {"⚔️ Farm", "🍎 Fruit", "⚡ Raid", "🎮 Misc", "📍 Teleport", "📊 Stats"}
local tabButtons = {}
local currentTab = 1

local tabFrame = Instance.new("Frame")
tabFrame.Size = UDim2.new(1, 0, 0, 50)
tabFrame.Position = UDim2.new(0, 0, 0, 40)
tabFrame.BackgroundTransparency = 1
tabFrame.Parent = main

for i, tabName in ipairs(tabs) do
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0, 90, 0, 35)
    btn.Position = UDim2.new(0, (i-1) * 95 + 10, 0, 7)
    btn.BackgroundColor3 = i == 1 and Color3.fromRGB(255, 200, 0) or Color3.fromRGB(40, 40, 40)
    btn.Text = tabName
    btn.TextColor3 = i == 1 and Color3.fromRGB(0, 0, 0) or Color3.fromRGB(255, 255, 255)
    btn.TextSize = 14
    btn.Font = Enum.Font.GothamBold
    btn.BorderSizePixel = 0
    btn.Parent = tabFrame
    
    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 6)
    btnCorner.Parent = btn
    
    tabButtons[i] = btn
    
    btn.MouseButton1Click:Connect(function()
        for j, b in ipairs(tabButtons) do
            b.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
            b.TextColor3 = Color3.fromRGB(255, 255, 255)
        end
        btn.BackgroundColor3 = Color3.fromRGB(255, 200, 0)
        btn.TextColor3 = Color3.fromRGB(0, 0, 0)
        currentTab = i
        updateContent()
    end)
end

-- Function to clear content
function clearContent()
    for _, child in ipairs(scrollFrame:GetChildren()) do
        child:Destroy()
    end
end

-- Function to create toggle
function createToggle(text, defaultY)
    local y = #scrollFrame:GetChildren() * 45
    
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, 0, 0, 40)
    frame.Position = UDim2.new(0, 0, 0, y)
    frame.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    frame.BackgroundTransparency = 0.3
    frame.Parent = scrollFrame
    
    local frameCorner = Instance.new("UICorner")
    frameCorner.CornerRadius = UDim.new(0, 6)
    frameCorner.Parent = frame
    
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, -50, 1, 0)
    label.Position = UDim2.new(0, 10, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = text
    label.TextColor3 = Color3.fromRGB(255, 255, 255)
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.Gotham
    label.TextSize = 14
    label.Parent = frame
    
    local toggle = Instance.new("TextButton")
    toggle.Size = UDim2.new(0, 30, 0, 30)
    toggle.Position = UDim2.new(1, -40, 0, 5)
    toggle.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
    toggle.Text = ""
    toggle.BorderSizePixel = 0
    toggle.Parent = frame
    
    local toggleCorner = Instance.new("UICorner")
    toggleCorner.CornerRadius = UDim.new(0, 15)
    toggleCorner.Parent = toggle
    
    local state = false
    
    toggle.MouseButton1Click:Connect(function()
        state = not state
        toggle.BackgroundColor3 = state and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(255, 0, 0)
    end)
    
    return frame
end

-- Function to create label

    local y = #scrollFrame:GetChildren() * 45
    
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, 0, 0, 30)
    label.Position = UDim2.new(0, 0, 0, y)
    label.BackgroundTransparency = 1
    label.Text = "▶ " .. text
    label.TextColor3 = Color3.fromRGB(255, 200, 0)
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.GothamBold

-- Update content based on current tab
function updateContent()
    clearContent()
    
    if currentTab == 1 then -- Farm
        createLabel("AUTO FARM")
        createToggle("Auto Farm")
        createToggle("Auto Quest")
        createToggle("Bring Mobs")
        createToggle("Auto Haki")
        createButton("Start Farming", function()
            game:GetService("StarterGui"):SetCore("SendNotification", {
                Title = "Delta Hub",
                Text = "Bắt đầu Auto Farm!",
                Duration = 2
            })
        end)
        
    elseif currentTab == 2 then -- Fruit
        createLabel("DEVIL FRUIT")
        createToggle("Auto Store Fruit")
        createToggle("Auto Eat Fruit")
        createToggle("Fruit Sniper")
        createButton("Find Fruit", function()
            game:GetService("StarterGui"):SetCore("SendNotification", {
                Title = "Delta Hub",
                Text = "Đang tìm fruit...",
                Duration = 2
            })
        end)
        
    elseif currentTab == 3 then -- Raid
        createLabel("RAID SETTINGS")
        createToggle("Auto Raid")
        createToggle("Auto Awakening")
        createButton("Start Raid", function()
            game:GetService("StarterGui"):SetCore("SendNotification", {
                Title = "Delta Hub",
                Text = "Bắt đầu Auto Raid!",
                Duration = 2
            })
        end)
        
    elseif currentTab == 4 then -- Misc
        createLabel("OTHER SETTINGS")
        createToggle("ESP")
        createToggle("Anti AFK")
        createButton("Refill HP", function()
            local args = {
                [1] = "RefillHP"
            }
            game:GetService("ReplicatedStorage").Remotes.CommF_:InvokeServer(unpack(args))
        end)
        
    elseif currentTab == 5 then -- Teleport
        createLabel("TELEPORT LOCATIONS")
        local locations = {
            "Jungle", "Desert", "Snow", "Marine", "Sky", "Mansion"
        }
        for _, loc in ipairs(locations) do
            createButton(loc, function()
                game:GetService("StarterGui"):SetCore("SendNotification", {
                    Title = "Delta Hub",
                    Text = "Đang teleport đến " .. loc,
                    Duration = 1
                })
            end)
        end
        
    elseif currentTab == 6 then -- Stats
        createLabel("PLAYER INFO")
        local info = Instance.new("TextLabel")
        info.Size = UDim2.new(1, 0, 0, 100)
        info.Position = UDim2.new(0, 0, 0, #scrollFrame:GetChildren() * 45)
        info.BackgroundTransparency = 1
        info.Text = "Name: " .. player.Name .. "\nLevel: 1000\nBeli: 1,000,000\nFragments: 5,000"
        info.TextColor3 = Color3.fromRGB(255, 255, 255)
        info.TextSize = 14
        info.Font = Enum.Font.Gotham
        info.Parent = scrollFrame
    end
    
    scrollFrame.CanvasSize = UDim2.new(0, 0, 0, #scrollFrame:GetChildren() * 45 + 50)
end

-- Initial update
updateContent()
