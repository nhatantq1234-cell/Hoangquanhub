local function CreateESP(instance)
    if not Config.EnableESP then return end

end

local startTime = os.time()
task.spawn(function()
    while true do
        local currentTime = os.time()
        local diff = currentTime - startTime
        local h = math.floor(diff / 3600)
        local m = math.floor((diff % 3600) / 60)
        local s = diff % 60
        TimeLabel.Text = string.format("Time : %dh %dm %ds", h, m, s)
        task.wait(1)
    end
end)

local currentTween = nil
local function TweenTo(targetCFrame)
    local HRP = GetRootPart()
    if not HRP then return end
    
    local Char = GetCharacter()
    local Hum = GetHumanoid()
    if not Hum or Hum.Health <= 0 then return end

    local Distance = (HRP.Position - targetCFrame.Position).Magnitude
    local Info = TweenInfo.new(Distance / Config.Speed, Enum.EasingStyle.Linear)
    
    if currentTween then currentTween:Cancel() end

    currentTween = TweenService:Create(HRP, Info, {CFrame = targetCFrame})
    currentTween:Play()
    
    local NoclipConn
    NoclipConn = RunService.Stepped:Connect(function()
        if currentTween and currentTween.PlaybackState == Enum.PlaybackState.Playing and Char then
            for _, part in pairs(Char:GetDescendants()) do
                if part:IsA("BasePart") and part.CanCollide then
                    part.CanCollide = false
                end
            end
            if HRP then HRP.Velocity = Vector3.zero end
        else
            if NoclipConn then NoclipConn:Disconnect() end
        end
    end)
    return currentTween
end

local function InvokeCommF(arg1, arg2, arg3)
    local CommF = ReplicatedStorage.Remotes.CommF_
    if CommF then
        return CommF:InvokeServer(arg1, arg2, arg3)
    end
end

local function StoreFruit(fruitInstance)
    local fruitName = fruitInstance:GetAttribute("OriginalName") or fruitInstance.Name
    local args = { [1] = "StoreFruit", [2] = fruitName, [3] = fruitInstance }
    pcall(function() ReplicatedStorage.Remotes.CommF_:InvokeServer(unpack(args)) end)
end

local function BuyRandomFruit()
    pcall(function() InvokeCommF("Cousin", "Buy") end)
end

local function AutoChooseTeam()
    pcall(function() InvokeCommF("SetTeam", Config.Team) end)
end

local function ServerHop()
    local PlaceID = game.PlaceId
    local AllIDs = {}
    local success, result = pcall(function()
        local record = HttpService:JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/" .. PlaceID .. "/servers/Public?sortOrder=Asc&limit=100"))
        for i, v in pairs(record.data) do
            if v.playing < v.maxPlayers and v.id ~= game.JobId then
                table.insert(AllIDs, v.id)
            end
        end
    end)
    
    if #AllIDs > 0 then
        TeleportService:TeleportToPlaceInstance(PlaceID, AllIDs[math.random(1, #AllIDs)], LocalPlayer)
    else
        UpdateStatus("Retrying Hop...")
        task.wait(1)
        ServerHop()
    end
end

-- Biến lưu thời gian bắt đầu đếm Hop
local HopStart = nil
-- Biến đếm số lần thử cất cho từng loại trái (Key = Fruit Name)
local StoredAttempts = {}
-- Biến đánh dấu đang bận cất (Để chặn Hop)
local IsBusyStoring = false

-- Tự động làm mới bộ đếm cất trái mỗi 8 giây
task.spawn(function()
    while true do
        task.wait(8)
        StoredAttempts = {} 
    end
end)

task.spawn(function()
    while true do
        task.wait()
        
        local Char = GetCharacter()
        local Hum = GetHumanoid()
        local HRP = GetRootPart()

        if Char and Hum and Hum.Health > 0 and HRP then
            
            IsBusyStoring = false

            -- GLOBAL AUTO STORE (Smart Limit + Fail Hop)
            if Config.AutoStore then
                local function tryStore(container)
                    for _, tool in pairs(container:GetChildren()) do
                        if tool:IsA("Tool") and (string.find(tool.Name, "Fruit") or tool:GetAttribute("OriginalName")) then
                            local fName = tool:GetAttribute("OriginalName") or tool.Name
                            if not StoredAttempts[fName] then StoredAttempts[fName] = 0 end
                            
                            -- Nếu vẫn còn lượt cất (< 5 lần)
                            if StoredAttempts[fName] < 5 then
                                IsBusyStoring = true -- Đánh dấu đang bận
                                UpdateStatus("Storing: " .. fName) -- Hiện status
                                StoreFruit(tool)
                                StoredAttempts[fName] = StoredAttempts[fName] + 1
                                task.wait(0.1) 
                            else
                                -- [MỚI] Nếu đã thử 5 lần mà vẫn còn trong túi -> Cất lỗi -> Hop Server
                                IsBusyStoring = true
                                UpdateStatus("Store Failed! Hopping...")
                                ServerHop()
                                task.wait(10) -- Chờ teleport
                            end
                        end
                    end
                end
                tryStore(LocalPlayer.Backpack)
                if Char then tryStore(Char) end
            end

            if Config.AutoBuyRandom then
                BuyRandomFruit()
            end

            local foundFruit = false
            local fruits = {}
            for _, v in pairs(Workspace:GetChildren()) do
                if (v:IsA("Tool") or v:IsA("Model")) and string.find(v.Name, "Fruit") then
                    if v:FindFirstChild("Handle") then
                        table.insert(fruits, v)
                    end
                end
            end

            for _, v in pairs(fruits) do
                foundFruit = true
                HopStart = nil
                
                CreateESP(v)
                UpdateStatus("Teleport " .. v.Name)
                
                local targetHandle = v:FindFirstChild("Handle")
                
                if targetHandle then
                    local tween = TweenTo(targetHandle.CFrame)
                    if tween then tween.Completed:Wait() end
                    
                    -- Anti-Despawn Logic
                    if v.Parent ~= Workspace then
                        local myFruit = LocalPlayer.Backpack:FindFirstChild(v.Name) or Char:FindFirstChild(v.Name)
                        if not myFruit then
                            UpdateStatus("Fruit Lost! Hopping...")
                            ServerHop() 
                            task.wait(10)
                        end
                    end

                    if (targetHandle.Position - HRP.Position).Magnitude < 15 then
                        if firetouchinterest then
                            firetouchinterest(HRP, targetHandle, 0)
                            firetouchinterest(HRP, targetHandle, 1)
                        end
                        local pickupTime = tick()
                        while v.Parent == Workspace and (tick() - pickupTime < 1) do
                            HRP.CFrame = targetHandle.CFrame
                            if Hum then Hum:EquipTool(v) end
                            task.wait()
                        end
                    end
                end
            end

            if not foundFruit then
                if currentTween then 
                    currentTween:Cancel() 
                    currentTween = nil
                end

                if Config.HopIfNoFruit then
                    -- NẾU ĐANG BẬN CẤT TRÁI HOẶC XỬ LÝ LỖI -> HỦY HOP BÌNH THƯỜNG
                    if IsBusyStoring then
                        HopStart = nil
                    else
                        if HopStart == nil then HopStart = tick() end
                        local timePassed = tick() - HopStart
                        local timeLeft = 7 - timePassed

                        if timeLeft > 0 then
                            UpdateStatus("Hop in " .. math.ceil(timeLeft) .. "s")
                        else
                            UpdateStatus("Hopping...")
                            ServerHop()
                            HopStart = nil 
                        end
                    end
                else
                    if not IsBusyStoring then
                        UpdateStatus("Scanning Area...") 
                    end
                    HopStart = nil
                end
            else
                HopStart = nil
            end
        else
            UpdateStatus("Waiting for Character...")
        end
    end
end)
