local function getChar()
    return lp.Character, lp.Character and lp.Character:FindFirstChild("HumanoidRootPart"), lp.Character and lp.Character:FindFirstChild("Humanoid")
end

local function teleport(pos)
    local _, root = getChar()
    if root then root.CFrame = CFrame.new(pos) end
end

-- Farm
spawn(function()
    while wait(0.1) do
        if cfg.farm then
            local nearest = nil
            local minDist = math.huge
            local _, root = getChar()
            if root then
                for _, mob in pairs(Workspace.Enemies:GetChildren()) do
                    local hum = mob:FindFirstChild("Humanoid")
                    local mRoot = mob:FindFirstChild("HumanoidRootPart")
                    if hum and hum.Health > 0 and mRoot then
                        local dist = (root.Position - mRoot.Position).magnitude
                        if dist < minDist and dist < 150 then
                            minDist = dist
                            nearest = mob
                        end
                    end
                end
                if nearest then
                    local mRoot = nearest:FindFirstChild("HumanoidRootPart")
                    if mRoot then
                        if (root.Position - mRoot.Position).magnitude > 15 then
                            root.CFrame = mRoot.CFrame * CFrame.new(0, 0, 5)
                        end
                        ReplicatedStorage.Remotes.CommF_:InvokeServer("DealDamage", mRoot.Position)
                    end
                end
            end
        end
        
        if cfg.bring then
            local _, root = getChar()
            if root then
                for _, mob in pairs(Workspace.Enemies:GetChildren()) do
                    local hum = mob:FindFirstChild("Humanoid")
                    local mRoot = mob:FindFirstChild("HumanoidRootPart")
                    if hum and hum.Health > 0 and mRoot and (root.Position - mRoot.Position).magnitude > 25 then
                        mRoot.CFrame = root.CFrame * CFrame.new(0, 0, 15)
                    end
                end
            end
        end
        
        if cfg.quest then
            local gui = lp.PlayerGui
            if gui and gui.Main and gui.Main.Quest and gui.Main.Quest.Visible == false then
                ReplicatedStorage.Remotes.CommF_:InvokeServer("StartQuest", "BanditQuest1", 1)
            end
        end
        
        if cfg.haki then
            ReplicatedStorage.Remotes.CommF_:InvokeServer("Buso")
        end
        
        if cfg.store then
            for _, fruit in pairs(Workspace:GetChildren()) do
                if fruit:IsA("Tool") and fruit:FindFirstChild("Handle") then
                    ReplicatedStorage.Remotes.CommF_:InvokeServer("StoreFruit", fruit.Name)
                    fruit:Destroy()
                end
            end
        end
        
        if cfg.snipe then
            for _, fruit in pairs(Workspace:GetChildren()) do
                if fruit:IsA("Tool") and fruit:FindFirstChild("Handle") then
                    if fruit.Name:lower():find("dragon") or fruit.Name:lower():find("leopard") then
                        teleport(fruit.Handle.Position)
                        wait(0.3)
                        fireproximityprompt(fruit:FindFirstChildOfClass("ProximityPrompt"), 0)
                    end
                end
            end
        end
        
        if cfg.raids then
            ReplicatedStorage.Remotes.CommF_:InvokeServer("Raids", "Select", "Flame")
            wait(1)
            ReplicatedStorage.Remotes.CommF_:InvokeServer("Raids", "Start")
            wait(300)
        end
    end
end)

-- Speed & Fly
spawn(function()
    while wait(0.1) do
        local char, root, hum = getChar()
        if cfg.speed and hum then
            hum.WalkSpeed = cfg.speedVal
            hum.JumpPower = cfg.jumpVal or 100
        end
        if cfg.fly and char then
            local bv = char:FindFirstChild("FlyBV")
            if not bv then
                bv = Instance.new("BodyVelocity")
                bv.Name = "FlyBV"
                bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
                bv.Parent = char
            end
            local dir = Vector3.new()
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + root.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - root.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - root.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + root.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0, 1, 0) end
            bv.Velocity = dir.Magnitude > 0 and dir.Unit * 50 or Vector3.new()
            if hum then hum.PlatformStand = true end
        elseif not cfg.fly and char and char:FindFirstChild("FlyBV") then
            char.FlyBV:Destroy()
            if hum then hum.PlatformStand = false end
        end
    end
end)
